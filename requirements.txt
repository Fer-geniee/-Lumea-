        perfil = cursor.fetchone() # 

        if not perfil:
            return jsonify({"error": "Usuario no encontrado"}), 404

        contraseña_valida = bcrypt.checkpw(contraseña.encode('utf-8'), 
        perfil['password_hash'].encode('utf-8'))
        if not contraseña_valida:
            return jsonify({"error": "Contraseña inválida"}), 401

        if contraseña_valida:
            del perfil['password_hash']  # No enviar el hash de contraseña al cliente
            return jsonify({"message": "Inicio de sesión exitoso", "perfil": perfil}), 200
        else:
            return jsonify({"error": "Contraseña inválida"}), 401
    except Error as e:
        print(f"Error al verificar inicio de sesión: {e}")
        return jsonify({"error": "Error interno del servidor"}), 500
    finally:
        cursor.close()
